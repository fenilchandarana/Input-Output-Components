#include <U8g2lib.h>
#include <Wire.h>

// U8g2 constructor for SH1106 I2C
U8G2_SH1106_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, /* reset=*/ U8X8_PIN_NONE);

void setup() {
  u8g2.begin();
}

void loop() {
  u8g2.clearBuffer();
  u8g2.setFont(u8g2_font_6x13_tr);  // similar to your old font
  u8g2.drawStr(0, 12, "row number 1");
  u8g2.drawStr(0, 26, "row number 2");
  u8g2.drawStr(0, 40, "row number 3");
  u8g2.sendBuffer();  // update the display
  delay(1000);
}
